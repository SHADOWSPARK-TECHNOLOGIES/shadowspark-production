.github/workflows/vercel-agent-sandbox.yml

Use this exact baseline, adapted only for package manager, Node version, monorepo layout, or Prisma generation if required.

```yaml
name: Vercel Agent Sandbox Review

on:
  pull_request:
    types:
      - opened
      - synchronize
      - reopened
      - ready_for_review
      - labeled
      - unlabeled

  issue_comment:
    types:
      - created

concurrency:
  group: vercel-agent-${{ github.event.issue.number || github.event.pull_request.number || github.run_id }}
  cancel-in-progress: true

permissions:
  contents: read

env:
  VERCEL_ORG_ID: ${{ secrets.VERCEL_ORG_ID }}
  VERCEL_PROJECT_ID: ${{ secrets.VERCEL_PROJECT_ID }}
  NODE_OPTIONS: "--max-old-space-size=4096"

jobs:
  route-vercel-command:
    name: Route @vercel command
    if: github.event_name == 'issue_comment' && github.event.issue.pull_request && contains(github.event.comment.body || '', '@vercel')
    runs-on: ubuntu-latest
    timeout-minutes: 5

    permissions:
      contents: read
      pull-requests: write

    steps:
      - name: Parse and authorize Vercel Agent command
        uses: actions/github-script@v7
        with:
          script: |
            const body = (context.payload.comment.body || '').trim();
            const issue_number = context.payload.issue.number;
            const actor = context.payload.comment.user.login;

            const { data: perm } = await github.rest.repos.getCollaboratorPermissionLevel({
              ...context.repo,
              username: actor,
            });

            const allowed = ['admin', 'maintain', 'write'].includes(perm.permission);

            if (!allowed) {
              await github.rest.issues.createComment({
                ...context.repo,
                issue_number,
                body: `@${actor} Vercel Agent commands require repository write access.`,
              });
              return;
            }

            const add = async (label) => {
              await github.rest.issues.addLabels({
                ...context.repo,
                issue_number,
                labels: [label],
              });
            };

            const remove = async (label) => {
              try {
                await github.rest.issues.removeLabel({
                  ...context.repo,
                  issue_number,
                  name: label,
                });
              } catch (e) {
                // Ignore missing label errors.
              }
            };

            const firstLine = body.split('\n')[0].trim();

            if (/^@vercel\s+(review|scan)\s*$/i.test(firstLine)) {
              await remove('agent:review-requested');
              await add('agent:review-requested');

              await github.rest.issues.createComment({
                ...context.repo,
                issue_number,
                body: 'Queued Vercel Agent read-only review and vulnerability scan.',
              });
            } else if (/^@vercel\s+deploy\s+preview\s+approve\s*$/i.test(firstLine)) {
              await remove('agent:deploy-preview-approved');
              await add('agent:deploy-preview-approved');

              await github.rest.issues.createComment({
                ...context.repo,
                issue_number,
                body: [
                  'Added `agent:deploy-preview-approved`.',
                  '',
                  'The protected `vercel-write` environment still requires manual reviewer sign-off before any write or deployment action.',
                ].join('\n'),
              });
            } else if (/^@vercel\s+reset\s*$/i.test(firstLine)) {
              await remove('agent:review-requested');
              await remove('agent:deploy-preview-approved');

              await github.rest.issues.createComment({
                ...context.repo,
                issue_number,
                body: 'Removed Vercel Agent request labels.',
              });
            } else {
              await github.rest.issues.createComment({
                ...context.repo,
                issue_number,
                body: [
                  'Supported Vercel Agent commands:',
                  '',
                  '- `@vercel review` — run read-only logic review and vulnerability scan',
                  '- `@vercel scan` — alias for `@vercel review`',
                  '- `@vercel deploy preview approve` — request protected preview deployment',
                  '- `@vercel reset` — remove Vercel Agent request labels',
                ].join('\n'),
              });
            }

  vercel-agent-review:
    name: Vercel Agent Read-only Review
    if: github.event_name == 'pull_request' && github.event.pull_request.draft == false
    runs-on: ubuntu-latest
    timeout-minutes: 25

    permissions:
      contents: read
      pull-requests: write

    steps:
      - name: Checkout PR merge ref
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm

      - name: Install dependencies
        run: npm ci --no-audit --no-fund

      - name: Logic checks
        id: logic
        continue-on-error: true
        run: |
          set -euo pipefail

          node - <<'NODE'
          const { execSync } = require('node:child_process');
          const fs = require('node:fs');

          if (!fs.existsSync('./package.json')) {
            console.log('No package.json found. Skipping logic checks.');
            process.exit(0);
          }

          const pkg = JSON.parse(fs.readFileSync('./package.json', 'utf8'));
          const scripts = pkg.scripts || {};
          const results = {};

          const run = (name) => {
            if (scripts[name]) {
              console.log(`::group::npm run ${name}`);
              execSync(`npm run ${name}`, { stdio: 'inherit' });
              console.log('::endgroup::');
              results[name] = 'passed';
              return true;
            }

            console.log(`Skipped missing script: ${name}`);
            results[name] = 'skipped';
            return false;
          };

          run('lint');
          run('typecheck') || run('tsc');
          run('test:ci') || run('test');

          fs.mkdirSync('.vercel-agent', { recursive: true });
          fs.writeFileSync('.vercel-agent/logic.json', JSON.stringify(results, null, 2));
          NODE

      - name: npm audit
        id: npm-audit
        continue-on-error: true
        run: npm audit --audit-level=high --omit=dev

      - name: Trivy filesystem scan
        id: trivy
        continue-on-error: true
        uses: aquasecurity/trivy-action@0.24.0
        with:
          scan-type: fs
          scan-ref: .
          scanners: vuln,secret,misconfig
          severity: HIGH,CRITICAL
          ignore-unfixed: true
          exit-code: '1'
          format: table

      - name: Post Vercel Agent review comment
        if: always()
        uses: actions/github-script@v7
        with:
          script: |
            const marker = '<!-- vercel-agent-read-only-review -->';
            const issue_number = context.payload.pull_request.number;

            const logic = '${{ steps.logic.outcome }}';
            const npmAudit = '${{ steps.npm-audit.outcome }}';
            const trivy = '${{ steps.trivy.outcome }}';

            const failed = [logic, npmAudit, trivy].includes('failure');
            const skippedAll = [logic, npmAudit, trivy].every((value) => value === 'skipped');
            const overall = failed ? 'failure' : skippedAll ? 'skipped' : 'success';

            const body = [
              marker,
              '## Vercel Agent read-only review',
              '',
              `- Overall status: **${overall}**`,
              `- Logic checks: **${logic}**`,
              `- npm audit: **${npmAudit}**`,
              `- Trivy scan: **${trivy}**`,
              '',
              'Mode: **read-only**.',
              '',
              'No write actions or deployments were performed.',
              '',
              'If this check failed, inspect the workflow logs for lint/type/test failures, npm audit findings, or Trivy findings.',
            ].join('\n');

            const { data: comments } = await github.rest.issues.listComments({
              ...context.repo,
              issue_number,
              per_page: 100,
            });

            const existing = comments.find((c) => c.body.includes(marker));

            if (existing) {
              await github.rest.issues.updateComment({
                ...context.repo,
                comment_id: existing.id,
                body,
              });
            } else {
              await github.rest.issues.createComment({
                ...context.repo,
                issue_number,
                body,
              });
            }

      - name: Fail if read-only checks failed
        if: always()
        run: |
          set -euo pipefail

          logic="${{ steps.logic.outcome }}"
          audit="${{ steps.npm-audit.outcome }}"
          trivy="${{ steps.trivy.outcome }}"

          if [[ "$logic" == "failure" || "$audit" == "failure" || "$trivy" == "failure" ]]; then
            echo "One or more Vercel Agent read-only checks failed."
            exit 1
          fi

  sandbox-build:
    name: Vercel Sandbox Build
    if: github.event_name == 'pull_request' && github.event.pull_request.draft == false
    runs-on: ubuntu-latest
    timeout-minutes: 30

    permissions:
      contents: read

    env:
      VERCEL_READ_TOKEN: ${{ secrets.VERCEL_READ_TOKEN }}

    steps:
      - name: Checkout PR merge ref
        uses: actions/checkout@v4

      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm

      - name: Install Vercel CLI
        run: npm install --global vercel@latest

      - name: Vercel pull preview environment
        if: env.VERCEL_READ_TOKEN != ''
        run: |
          vercel pull \
            --yes \
            --environment=preview \
            --git-branch="${{ github.head_ref }}" \
            --token="$VERCEL_READ_TOKEN"

      - name: Vercel sandbox build
        if: env.VERCEL_READ_TOKEN != ''
        run: |
          vercel build \
            --token="$VERCEL_READ_TOKEN"

      - name: Fallback local build
        if: env.VERCEL_READ_TOKEN == ''
        run: |
          npm ci --no-audit --no-fund
          npm run build --if-present

      - name: Upload sandbox build artifact
        uses: actions/upload-artifact@v4
        with:
          name: vercel-sandbox-output
          path: |
            .vercel/output
            .next
            dist
            build
          if-no-files-found: ignore
          retention-days: 7

  deploy-preview-approved:
    name: Vercel Preview Deploy (write)
    if: >
      github.event_name == 'pull_request' &&
      github.event.pull_request.draft == false &&
      github.event.pull_request.head.repo.full_name == github.repository &&
      contains(github.event.pull_request.labels.*.name, 'agent:deploy-preview-approved')
    needs:
      - vercel-agent-review
      - sandbox-build
    runs-on: ubuntu-latest
    timeout-minutes: 30

    environment: vercel-write

    permissions:
      contents: read
      pull-requests: write

    env:
      VERCEL_DEPLOY_TOKEN: ${{ secrets.VERCEL_DEPLOY_TOKEN }}

    steps:
      - name: Checkout PR merge ref
        uses: actions/checkout@v4

      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm

      - name: Install Vercel CLI
        run: npm install --global vercel@latest

      - name: Vercel pull preview environment
        run: |
          vercel pull \
            --yes \
            --environment=preview \
            --git-branch="${{ github.head_ref }}" \
            --token="$VERCEL_DEPLOY_TOKEN"

      - name: Vercel build
        run: |
          vercel build \
            --token="$VERCEL_DEPLOY_TOKEN"

      - name: Vercel deploy preview
        id: deploy
        run: |
          set -euo pipefail

          url="$(vercel deploy --prebuilt --yes --token="$VERCEL_DEPLOY_TOKEN" | tail -n 1)"

          echo "url=$url" >> "$GITHUB_OUTPUT"

      - name: Post deployment comment
        if: always()
        uses: actions/github-script@v7
        with:
          script: |
            const marker = '<!-- vercel-agent-preview-deploy -->';
            const issue_number = context.payload.pull_request.number;

            const status = '${{ job.status }}';
            const url = '${{ steps.deploy.outputs.url }}';

            const body = [
              marker,
              '## Vercel Agent preview deployment',
              '',
              `- Status: **${status}**`,
              url ? `- URL: ${url}` : '- URL: unavailable, check workflow logs',
              '',
              'This deployment used the protected `vercel-write` environment after manual sign-off.',
            ].join('\n');

            const { data: comments } = await github.rest.issues.listComments({
              ...context.repo,
              issue_number,
              per_page: 100,
            });

            const existing = comments.find((c) => c.body.includes(marker));

            if (existing) {
              await github.rest.issues.updateComment({
                ...context.repo,
                comment_id: existing.id,
                body,
              });
            } else {
              await github.rest.issues.createComment({
                ...context.repo,
                issue_number,
                body,
              });
            }

# Vercel Agent Rules

This repository uses Vercel Agent in a strict, read-only-by-default configuration.

## Default mode

- Operate in read-only mode by default.
- Do not modify source code unless explicitly authorized through a pull request workflow.
- Do not push directly to protected branches.
- Do not open pull requests unless authorized.
- Do not change repository settings.
- Do not change Vercel project settings.
- Do not modify environment variables.
- Do not access production secrets.
- Do not deploy to production.

## Allowed read-only actions

You may:

- Read pull request diffs.
- Read repository code at the pull request merge ref.
- Run lint, typecheck, and test commands in sandboxed CI.
- Run vulnerability scans in sandboxed CI.
- Run `vercel pull` for preview environment metadata using read-only credentials.
- Run `vercel build` for sandbox build validation.
- Post summary comments on pull requests.

## Vulnerability scanning

When asked to review a pull request:

- Run logic checks.
- Run dependency vulnerability scanning.
- Run secret scanning.
- Run misconfiguration scanning where applicable.
- Fail the review for high or critical findings unless a human security owner has explicitly approved an exception.

## Sandbox build validation

Before a pull request can merge:

- A sandbox build must pass.
- The sandbox build must use preview environment configuration.
- The sandbox build must not deploy to production.
- Build artifacts must be ephemeral.
- Secrets must be exposed only as read-only preview environment variables.

## Write actions and deployments

Write actions require scoped manual approval.

A write action is allowed only when all of the following are true:

1. The pull request has the label `agent:deploy-preview-approved`.
2. The read-only review check has passed.
3. The sandbox build check has passed.
4. A human reviewer approves the protected GitHub Environment `vercel-write`.

Even after approval:

- Only preview deployment is allowed.
- Production deployment is denied.
- Repository write access is denied.
- Environment variable modification is denied.
- Vercel project settings modification is denied.

## Commands

Supported pull request commands:

```text
@vercel review
@vercel scan
@vercel deploy preview approve
@vercel reset
```

`@vercel review` and `@vercel scan` are read-only.

`@vercel deploy preview approve` requests a protected preview deployment, but it still requires manual GitHub Environment approval.

`@vercel reset` removes Agent request labels.
